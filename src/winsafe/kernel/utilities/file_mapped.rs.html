4/// Manages an [`HFILEMAP`](crate::HFILEMAP) handle, which provides
5/// memory-mapped file operations, including read/write through slices. It is
6/// closed automatically when the object goes out of scope.
7///
8/// # Examples
9///
10/// Usually the fastest way to read a file as string is by memory-mapping it,
11/// like:
12///
13/// ```no_run
14/// use winsafe::{self as w, prelude::*};
15///
16/// let f = w::FileMapped::open(
17///     "C:\\Temp\\foo.txt",
18///     w::FileAccess::ExistingReadOnly,
19/// )?;
20/// let raw_bytes = f.as_slice();
21/// let text = w::WString::parse(raw_bytes)?.to_string();
22/// # w::SysResult::Ok(())
23/// ```
24pub struct FileMapped {
25	hview: UnmapViewOfFileGuard, // drop order is important
26	_hmap: CloseHandleGuard<HFILEMAP>,
27	file: File,
28	size: u64,
29}
30
31impl FileMapped {
32	/// Opens a file with the desired access, then map its contents in memory.
33	#[must_use]
34	pub fn open(file_path: &str, access: FileAccess) -> SysResult<Self> {
35		let file = File::open(file_path, access)?;
36		let hmap = file.hfile().CreateFileMapping(
37			None,
38			match access {
39				FileAccess::ExistingReadOnly => co::PAGE::READONLY,
40				FileAccess::ExistingRW | FileAccess::OpenOrCreateRW | FileAccess::CreateRW => {
41					co::PAGE::READWRITE
42				},
43			},
44			None,
45			None,
46			None,
47		)?;
48		let hview = hmap.MapViewOfFile(
49			match access {
50				FileAccess::ExistingReadOnly => co::FILE_MAP::READ,
51				FileAccess::ExistingRW | FileAccess::OpenOrCreateRW | FileAccess::CreateRW => {
52					co::FILE_MAP::READ | co::FILE_MAP::WRITE
53				},
54			},
55			0,
56			None,
57		)?;
58		let size = file.hfile().GetFileSizeEx()?; // cache
59		Ok(Self { file, _hmap: hmap, hview, size })
60	}
61
62	/// Returns a mutable slice to the mapped memory.
63	#[must_use]
64	pub fn as_mut_slice(&mut self) -> &mut [u8] {
65		self.hview.as_mut_slice(self.size as _)
66	}
67
68	/// Returns a slice to the mapped memory.
69	#[must_use]
70	pub fn as_slice(&self) -> &[u8] {
71		self.hview.as_slice(self.size as _)
72	}
73
74	/// Returns the underlying file handle.
75	#[must_use]
76	pub fn hfile(&self) -> &HFILE {
77		self.file.hfile()
78	}
79
80	/// Returns the size of the file.
81	///
82	/// This value is cached.
83	#[must_use]
84	pub const fn size(&self) -> u64 {
85		self.size
86	}
87
88	/// Returns, in current time zone, 3 times of the file, respectively:
89	/// 1. creation time;
90	/// 2. last access time;
91	/// 3. last write time.
92	#[must_use]
93	pub fn times(&self) -> SysResult<(SYSTEMTIME, SYSTEMTIME, SYSTEMTIME)> {
94		self.file.times()
95	}
96}

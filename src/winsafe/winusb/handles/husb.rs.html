57	/// [`WinUsb_GetDescriptor`](https://learn.microsoft.com/en-us/windows/win32/api/winusb/nf-winusb-winusb_getdescriptor)
58	/// function.
59	///
60	/// Note that, currently, not all descriptors are implemented. You can query
61	/// only those defined in [`UsbDescr`](crate::UsbDescr) enum, otherwise this
62	/// method will return
63	/// [`ERROR::INVALID_PARAMETER`](crate::co::ERROR::INVALID_PARAMETER).
64	#[must_use]
65	pub fn GetDescriptor(
66		&self,
67		descriptor_type: co::USB_DESCRIPTOR_TYPE,
68		index: u8,
69		lang_id: LANGID,
70	) -> SysResult<UsbDescr> {
71		let mut bytes_read = 0u32;
72		match descriptor_type {
73			co::USB_DESCRIPTOR_TYPE::DEVICE => {
74				let mut obj = USB_DEVICE_DESCRIPTOR::default();
75				let ret = BoolRet(unsafe {
76					ffi::WinUsb_GetDescriptor(
77						self.ptr(),
78						descriptor_type.raw(),
79						index,
80						lang_id.raw(),
81						&mut obj as *mut _ as _,
82						std::mem::size_of::<USB_DEVICE_DESCRIPTOR>() as _,
83						&mut bytes_read,
84					)
85				})
86				.to_sysresult()
87				.map(|_| UsbDescr::Device(obj));
88
89				if bytes_read == std::mem::size_of::<USB_DEVICE_DESCRIPTOR>() as _ {
90					ret
91				} else {
92					Err(co::ERROR::INCORRECT_SIZE)
93				}
94			},
95			co::USB_DESCRIPTOR_TYPE::CONFIGURATION => {
96				let mut obj = USB_CONFIGURATION_DESCRIPTOR::default();
97				let ret = BoolRet(unsafe {
98					ffi::WinUsb_GetDescriptor(
99						self.ptr(),
100						descriptor_type.raw(),
101						index,
102						lang_id.raw(),
103						&mut obj as *mut _ as _,
104						std::mem::size_of::<USB_CONFIGURATION_DESCRIPTOR>() as _,
105						&mut bytes_read,
106					)
107				})
108				.to_sysresult();
109
110				if ret.is_err() {
111					return Err(ret.unwrap_err());
112				} else if bytes_read != std::mem::size_of::<USB_CONFIGURATION_DESCRIPTOR>() as _ {
113					return Err(co::ERROR::INCORRECT_SIZE);
114				}
115
116				let mut obj2 = UsbConfiguratorDescriptorGuard::new(obj.wTotalLength as _);
117				let ret2 = BoolRet(unsafe {
118					ffi::WinUsb_GetDescriptor(
119						self.ptr(),
120						descriptor_type.raw(),
121						index,
122						lang_id.raw(),
123						obj2.as_mut_ptr(),
124						obj.wTotalLength as _,
125						&mut bytes_read,
126					)
127				})
128				.to_sysresult()
129				.map(|_| UsbDescr::Configuration(obj2));
130
131				if bytes_read == std::mem::size_of::<USB_DEVICE_DESCRIPTOR>() as _ {
132					ret2
133				} else {
134					Err(co::ERROR::INCORRECT_SIZE)
135				}
136			},
137			co::USB_DESCRIPTOR_TYPE::STRING => {
138				let mut buf = vec![0u8; 256]; // max size of USB descriptor
139				let ret = BoolRet(unsafe {
140					ffi::WinUsb_GetDescriptor(
141						self.ptr(),
142						descriptor_type.raw(),
143						index,
144						lang_id.raw(),
145						buf.as_mut_ptr(),
146						buf.len() as _,
147						&mut bytes_read,
148					)
149				})
150				.to_sysresult();
151
152				// USB_STRING_DESCRIPTOR layout:
153				//
154				// Offset 0: bLength
155				// Offset 1: bDescriptorType
156				// Offset 2: UTF-16LE string data
157
158				if ret.is_err() {
159					return Err(ret.unwrap_err());
160				} else if bytes_read < 2 || buf[1] != co::USB_DESCRIPTOR_TYPE::STRING.raw() {
161					return Err(co::ERROR::INVALID_DATA);
162				}
163
164				let str_len = buf[0] as u32;
165				if str_len < 2 || str_len > bytes_read || (str_len % 2) != 0 {
166					return Err(co::ERROR::INVALID_DATA);
167				}
168
169				Ok(UsbDescr::String(
170					WString::from_wchars_count(buf.as_ptr() as _, (str_len as usize - 2) / 2)
171						.to_string(),
172				))
173			},
174			_ => Err(co::ERROR::INVALID_PARAMETER),
175		}
176	}

218	/// [`WinUsb_ReadPipe`](https://learn.microsoft.com/en-us/windows/win32/api/winusb/nf-winusb-winusb_readpipe)
219	/// function.
220	///
221	/// Returns the number of bytes read.
222	///
223	/// Note that asynchronous reading – which use the
224	/// [`OVERLAPPED`](crate::OVERLAPPED) struct – is not currently supported by
225	/// this method, because the buffer must remain untouched until the async
226	/// operation is complete, thus making the method unsound.
227	pub fn ReadPipe(&self, pipe_id: u8, buffer: &mut [u8]) -> SysResult<u32> {
228		let mut bytes_read = 0u32;
229		BoolRet(unsafe {
230			ffi::WinUsb_ReadPipe(
231				self.ptr(),
232				pipe_id,
233				buffer.as_mut_ptr(),
234				buffer.len() as _,
235				&mut bytes_read,
236				std::ptr::null_mut(),
237			)
238		})
239		.to_sysresult()
240		.map(|_| bytes_read)
241	}

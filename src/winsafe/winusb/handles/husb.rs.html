57	/// [`WinUsb_GetDescriptor`](https://learn.microsoft.com/en-us/windows/win32/api/winusb/nf-winusb-winusb_getdescriptor)
58	/// function.
59	///
60	/// Note that, currently, not all descriptors are implemented. You can query
61	/// only those defined in [`UsbDescr`](crate::UsbDescr) enum, otherwise this
62	/// method will return
63	/// [`ERROR::INVALID_PARAMETER`](crate::co::ERROR::INVALID_PARAMETER).
64	#[must_use]
65	pub fn GetDescriptor(
66		&self,
67		descriptor_type: co::USB_DESCRIPTOR_TYPE,
68		index: u8,
69		lang_id: LANGID,
70	) -> SysResult<UsbDescr> {
71		let mut bytes_read = 0u32;
72		match descriptor_type {
73			co::USB_DESCRIPTOR_TYPE::DEVICE => {
74				let mut obj = USB_DEVICE_DESCRIPTOR::default();
75				let ret = BoolRet(unsafe {
76					ffi::WinUsb_GetDescriptor(
77						self.ptr(),
78						descriptor_type.raw(),
79						index,
80						lang_id.raw(),
81						&mut obj as *mut _ as _,
82						std::mem::size_of::<USB_DEVICE_DESCRIPTOR>() as _,
83						&mut bytes_read,
84					)
85				})
86				.to_sysresult()
87				.map(|_| UsbDescr::Device(obj));
88
89				if bytes_read == std::mem::size_of::<USB_DEVICE_DESCRIPTOR>() as _ {
90					ret
91				} else {
92					Err(co::ERROR::INCORRECT_SIZE)
93				}
94			},
95			co::USB_DESCRIPTOR_TYPE::CONFIGURATION => {
96				let mut obj = USB_CONFIGURATION_DESCRIPTOR::default();
97				let ret = BoolRet(unsafe {
98					ffi::WinUsb_GetDescriptor(
99						self.ptr(),
100						descriptor_type.raw(),
101						index,
102						lang_id.raw(),
103						&mut obj as *mut _ as _,
104						std::mem::size_of::<USB_CONFIGURATION_DESCRIPTOR>() as _,
105						&mut bytes_read,
106					)
107				})
108				.to_sysresult();
109
110				if ret.is_err() {
111					return Err(ret.unwrap_err());
112				} else if bytes_read != std::mem::size_of::<USB_CONFIGURATION_DESCRIPTOR>() as _ {
113					return Err(co::ERROR::INCORRECT_SIZE);
114				}
115
116				let mut obj2 = UsbConfiguratorDescriptorGuard::new(obj.wTotalLength as _);
117				let ret2 = BoolRet(unsafe {
118					ffi::WinUsb_GetDescriptor(
119						self.ptr(),
120						descriptor_type.raw(),
121						index,
122						lang_id.raw(),
123						obj2.as_mut_ptr(),
124						obj.wTotalLength as _,
125						&mut bytes_read,
126					)
127				})
128				.to_sysresult()
129				.map(|_| UsbDescr::Configuration(obj2));
130
131				if bytes_read == std::mem::size_of::<USB_DEVICE_DESCRIPTOR>() as _ {
132					ret2
133				} else {
134					Err(co::ERROR::INCORRECT_SIZE)
135				}
136			},
137			_ => Err(co::ERROR::INVALID_PARAMETER),
138		}
139	}

3use crate::decl::*;
4use crate::kernel::privs::*;
5use crate::macros::*;

8handle! { HUSB;
9	/// Handles to an
10	/// [USB device](https://learn.microsoft.com/en-us/windows/win32/api/winusb/nf-winusb-winusb_initialize).
11	/// Originally `WINUSB_INTERFACE_HANDLE`.
12	///
13	/// Usually created with
14	/// [`HFILE::WinUsbInitialize`](crate::HFILE::WinUsbInitialize) method.
15}
16
17impl HUSB {
18	/// [`WinUsb_AbortPipe`](https://learn.microsoft.com/en-us/windows/win32/api/winusb/nf-winusb-winusb_abortpipe)
19	/// function.
20	pub fn AbortPipe(&self, pipe_id: u8) -> SysResult<()> {
21		BoolRet(unsafe { ffi::WinUsb_AbortPipe(self.ptr(), pipe_id) }).to_sysresult()
22	}
23
24	/// [`WinUsb_FlushPipe`](https://learn.microsoft.com/en-us/windows/win32/api/winusb/nf-winusb-winusb_flushpipe)
25	/// function.
26	pub fn FlushPipe(&self, pipe_id: u8) -> SysResult<()> {
27		BoolRet(unsafe { ffi::WinUsb_FlushPipe(self.ptr(), pipe_id) }).to_sysresult()
28	}
29
30	/// [`WinUsb_GetCurrentAlternateSetting`](https://learn.microsoft.com/en-us/windows/win32/api/winusb/nf-winusb-winusb_getcurrentalternatesetting)
31	/// function.
32	#[must_use]
33	pub fn GetCurrentAlternateSetting(&self) -> SysResult<u8> {
34		let mut setting_no = 0u8;
35		BoolRet(unsafe { ffi::WinUsb_GetCurrentAlternateSetting(self.ptr(), &mut setting_no) })
36			.to_sysresult()
37			.map(|_| setting_no)
38	}
39
40	/// [`WinUsb_GetCurrentFrameNumber`](https://learn.microsoft.com/en-us/windows/win32/api/winusb/nf-winusb-winusb_getcurrentframenumber)
41	/// function.
42	///
43	/// Returns the current frame number and the timestamp, respectively.
44	#[must_use]
45	pub fn GetCurrentFrameNumber(&self) -> SysResult<(u32, i64)> {
46		let mut current_frame_no = 0u32;
47		let mut timestamp = 0i64;
48
49		BoolRet(unsafe {
50			ffi::WinUsb_GetCurrentFrameNumber(self.ptr(), &mut current_frame_no, &mut timestamp)
51		})
52		.to_sysresult()
53		.map(|_| (current_frame_no, timestamp))
54	}
55
56	/// [`WinUsb_ResetPipe`](https://learn.microsoft.com/en-us/windows/win32/api/winusb/nf-winusb-winusb_resetpipe)
57	/// function.
58	pub fn ResetPipe(&self, pipe_id: u8) -> SysResult<()> {
59		BoolRet(unsafe { ffi::WinUsb_ResetPipe(self.ptr(), pipe_id) }).to_sysresult()
60	}
61
62	/// [`WinUsb_SetCurrentAlternateSetting`](https://learn.microsoft.com/en-us/windows/win32/api/winusb/nf-winusb-winusb_setcurrentalternatesetting)
63	/// function.
64	pub fn SetCurrentAlternateSetting(&self, setting_no: u8) -> SysResult<()> {
65		BoolRet(unsafe { ffi::WinUsb_SetCurrentAlternateSetting(self.ptr(), setting_no) })
66			.to_sysresult()
67	}
68}
